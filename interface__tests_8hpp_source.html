21// Test that clock of chip works correctly:
22// - time increases
23struct clock_test : ftest::utest
24{
25 clk_iface& clk;
26
27 char const* name = "clock_test";
28
29 ftest::task< void > exec()
30 {
31 co_await init_utest( *this );
32
33 std::size_t wait_cycles = 100;
34 microseconds t1 = clk.get_us();
35 for ( std::size_t i = 0; i < wait_cycles; i++ )
36 asm( "nop" );
37 microseconds t2 = clk.get_us();
38 co_await expect( *this, t2 > t1 );
39 co_await store_metric( *this, "t1", t1 );
40 co_await store_metric( *this, "t2", t2 );
41 co_await store_metric( *this, "wait_cycles", wait_cycles );
42
43 microseconds now = clk.get_us();
44 microseconds last = now;
45 microseconds end = now + 2_s;
46 do {
47 last = now;
48 now = clk.get_us();
49 if ( last > now ) {
50 co_await store_metric( *this, "last", last );
51 co_await store_metric( *this, "now", now );
52 }
53 co_await expect( *this, last <= now );
54 } while ( clk.get_us() < end );
55 }
56};
