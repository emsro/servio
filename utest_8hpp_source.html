445template < typename T >
446void setup_utest( asrt::task_ctx& ctx, asrt_reac_assm& assm, status& res, T def )
447{
448 if ( res != status::success )
449 return;
450
451 using U = asrt::task_unit< T >;
452 auto& a = ctx.query( ecor::get_memory_resource );
453 void* p = a.allocate( sizeof( U ), alignof( U ) );
454 if ( !p ) {
455 res = status::error;
456 return;
457 }
458
459 auto& test = *new ( p ) U{ T{ std::move( def ) } };
460 if ( asrt::add_test( assm.reactor, test ) != ASRT_SUCCESS ) {
461 res = status::error;
462 return;
463 }
464}
